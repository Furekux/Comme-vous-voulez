SELECT Activite.nom_activite, count(Inscription.id_activite) AS 'nb_inscription'
FROM Inscription 
INNER JOIN Activite ON Activite.id_activite = Inscription.id_activite
GROUP BY Activite.nom_activite
HAVING nb_inscription > 1
ORDER BY nb_inscription
